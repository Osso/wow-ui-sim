+"""Retain a streamed command log and receipt; --start launches without waiting."""
+import hashlib
+import json
+import os
+from pathlib import Path
+import subprocess
+import sys
+import time
+
+ROOT = Path(__file__).resolve().parents[4]
+EVIDENCE = Path(__file__).resolve().parent
+TARGET = '/home/osso/.cache/wow-ui-sim-targets/p540-page'
+
+
+def run_proof(name, command):
+    revision = subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=ROOT, text=True).strip()
+    started = time.monotonic()
+    log = EVIDENCE / f'{name}.txt'
+    environment = {key: value for key, value in os.environ.items()
+                   if key.startswith(('P540_', 'PATCH_', 'WOW_SIM_', 'PREFORK_'))
+                   or key.endswith('_SWEEP_OUT')}
+    with log.open('w') as output:
+        result = subprocess.run(command, cwd=ROOT,
+                                env=dict(os.environ, CARGO_TARGET_DIR=TARGET,
+                                         PYTHONDONTWRITEBYTECODE='1'),
+                                stdout=output, stderr=subprocess.STDOUT)
+    receipt = {'command': command, 'revision': revision, 'scope': name,
+               'cwd': '.', 'target': 'dedicated p540-page target (external to repository)',
+               'environment': environment, 'exit': result.returncode,
+               'log': log.name, 'log_sha256': hashlib.sha256(log.read_bytes()).hexdigest(),
+               'seconds': time.monotonic() - started, 'invalidated': False}
+    (EVIDENCE / f'{name}.proof.json').write_text(json.dumps(receipt, indent=2) + '\n')
+    print(name, result.returncode)
+    return result.returncode
+
+
+if __name__ == '__main__':
+    arguments = sys.argv[1:]
+    if arguments[0] == '--start':
+        name, *command = arguments[1:]
+        with (EVIDENCE / f'{name}.launcher.txt').open('w') as output:
+            process = subprocess.Popen([sys.executable, str(Path(__file__).resolve()), name, *command],
+                                       cwd=ROOT, stdout=output, stderr=subprocess.STDOUT,
+                                       start_new_session=True)
+        print(json.dumps({'name': name, 'pid': process.pid, 'log': f'{name}.txt'}))
+    else:
+        sys.exit(run_proof(arguments[0], arguments[1:]))

+                'Events': 'events', 'Widget API': 'widgets'}

+"""2013 Mists summary and separately pinned diff contracts."""
+import unittest
+import gen_patch_wikitext_register as generator

+
+class MistsTests(unittest.TestCase):
+    def test_caption_inventory_keeps_bare_removal_and_counts(self):
+        raw = ('{| class="wikitable"\n|+ Global API (5.3 to 5.4)\n'
+               '! style="width:50%"| 1 new functions\n'
+               '! style="width:50%"| 1 removed functions\n'
+               '| valign="top" | <div>\n: {{api|C_NewItems.ClearAll}}\n</div>\n'
+               '| valign="top" | <div>\n: AcceptArenaTeam\n</div>\n|}\n')
+        rows, counts = generator.parse_mists_automated_diff(raw)
+        self.assertEqual([(r['symbol'], r['direction'], r['wikitext_line']) for r in rows],
+                         [('C_NewItems.ClearAll', 'added', 6), ('AcceptArenaTeam', 'removed', 9)])
+        self.assertEqual([(r['header_count'], r['parsed_count']) for r in counts], [(1, 1), (1, 1)])
+
+    def test_summary_preserves_occurrences_and_reference_kinds(self):
+        raw = ('==Breaking changes==\n* {{api|GetAuctionItemInfo}} returns changed\n'
+               '==Modified API==\n* {{api|GetInstanceInfo}}()\n'
+               '==Bugs==\n* {{api|t=w|Slider:SetValueStep|valueStep}} ignored\n'
+               '==New features==\n* {{api|UnitRealmRelationship}}\n'
+               '* {{api|t=e|SAVED_VARIABLES_TOO_LARGE}}\n'
+               '==Automated diff==\n{{:Patch 5.4.0/API changes/diff}}\n')
+        rows = generator.parse_mists_summary(raw)
+        self.assertEqual([(r['section'], r['symbol'], r['direction']) for r in rows], [
+            ('global-api', 'GetAuctionItemInfo', 'changed'),
+            ('global-api', 'GetInstanceInfo', 'changed'),
+            ('widgets', 'Slider:SetValueStep', 'changed'),
+            ('global-api', 'UnitRealmRelationship', 'added'),
+            ('events', 'SAVED_VARIABLES_TOO_LARGE', 'added')])
+        self.assertTrue(all(r['annotation'] for r in rows))
